#!/usr/bin/env python3 """ Scraper di notizie reali per "Luisa Spagnoli" (retail/export intelligence). Come funziona: - Interroga i feed RSS di Google News (nessuna API key richiesta) su più combinazioni lingua/paese, cercando notizie che citano "Luisa Spagnoli" o temi correlati (retail, export, dazi, mercato moda) per ciascun mercato. - Deduplica per URL/titolo, ordina per data decrescente. - Salva tutto in data.json, pronto da incollare nella dashboard HTML (variabile `articles`) o da servire con un piccolo backend. Uso: pip install feedparser python3 scraper_luisa_spagnoli.py Nota: uno scraper "vivo" non può girare dentro una pagina HTML pubblicata (i browser bloccano richieste a domini esterni non autorizzati per motivi di sicurezza). Va eseguito qui, come script, e poi il risultato va caricato nella dashboard. """ import feedparser import json import hashlib from datetime import datetime, timezone from urllib.parse import quote # Mercato -> lista di (query, lingua-paese Google News) MARKETS = { "ITA": [ ('"Luisa Spagnoli"', "it-IT"), ('"Luisa Spagnoli" retail OR sfilata OR export', "it-IT"), ], "EUR": [ ('"Luisa Spagnoli"', "fr-FR"), ('"Luisa Spagnoli"', "de-DE"), ('"Luisa Spagnoli"', "en-GB"), ('"Luisa Spagnoli"', "es-ES"), ], "RUS": [ ('"Luisa Spagnoli"', "ru-RU"), ], "USA": [ ('"Luisa Spagnoli"', "en-US"), ], } # hl = lingua interfaccia, gl = paese, ceid = country:lang per il feed LANG_MAP = { "it-IT": ("it", "IT"), "fr-FR": ("fr", "FR"), "de-DE": ("de", "DE"), "en-GB": ("en", "GB"), "es-ES": ("es", "ES"), "ru-RU": ("ru", "RU"), "en-US": ("en", "US"), } def build_feed_url(query: str, locale: str) -> str: hl, gl = LANG_MAP[locale] q = quote(query) return f"https://news.google.com/rss/search?q={q}&hl={hl}&gl={gl}&ceid={gl}:{hl}" def fetch_market(market: str, queries): results = [] for query, locale in queries: url = build_feed_url(query, locale) feed = feedparser.parse(url) for entry in feed.entries: title = getattr(entry, "title", "").strip() link = getattr(entry, "link", "").strip() if not title or not link: continue source = "" if hasattr(entry, "source") and hasattr(entry.source, "title"): source = entry.source.title published = getattr(entry, "published", "") try: dt = datetime(*entry.published_parsed[:6], tzinfo=timezone.utc) date_str = dt.strftime("%Y-%m-%d") except Exception: date_str = published[:10] if published else "" summary = getattr(entry, "summary", "") uid = hashlib.md5(link.encode("utf-8")).hexdigest()[:10] results.append({ "id": uid, "market": market, "source": source or "Google News", "title": title, "url": link, "date": date_str, "preview": summary[:220], }) return results def main(): all_articles = [] seen_urls = set() for market, queries in MARKETS.items(): print(f"Scraping mercato {market}...") for art in fetch_market(market, queries): if art["url"] in seen_urls: continue seen_urls.add(art["url"]) all_articles.append(art) all_articles.sort(key=lambda a: a["date"], reverse=True) with open("data.json", "w", encoding="utf-8") as f: json.dump(all_articles, f, ensure_ascii=False, indent=2) print(f"\nSalvate {len(all_articles)} notizie uniche in data.json") if __name__ == "__main__": main()
